# Edit Deck Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a DECK tab to the lobby where players set copies per card (with odds), edit built-in cards and add custom cards, all saved on the device.

**Architecture:** A pure `src/lib/deck.js` turns `cards.json` + a small "deck edits" object into display rows (copies, odds, status) and the expanded deck the engine deals from. `useDeck` persists the edits in `localStorage`. `DeckPanel` + `CardEditor` render it inside a tabbed `LobbyScreen`; `useGame.nextCard(cards)` deals from the built deck.

**Tech Stack:** React 19, Vite 8 beta, plain CSS (Neon Street tokens), Node 22 `node --test`.

Spec: `docs/superpowers/specs/2026-10-20-edit-deck-design.md`

## Global Constraints

- Spacing only `--s1..--s5` (8/16/24/40/64px); colours only Neon Street tokens; fonts as in the Neon Street spec.
- Copies per card: integer 0–5, default 1; 0 = off.
- Odds = copies ÷ total copies; format `—` (0), `<0.1%`, one decimal under 10%, whole number otherwise.
- Title ≤ 40 chars, description ≤ 160 chars, both required.
- Custom ids are strings `c-…`; built-in ids stay numbers. `cards.json` is never modified.
- Storage key `drunkgamer:deck`; no new dependencies; all motion respects reduced motion.
- Run commands from `drunkgamerfront/`.

// Deck editing: combines the built-in cards with this device's edits
// (copies per card, reworded built-ins, custom cards). Pure functions.

const RULE = /^(person|group)_round_based:(\d+)$/;

export const MAX_COPIES = 5;
export const KINDS = ["chaos", "dare", "truth", "opinion", "power"];
export const LIMITS = { title: 40, description: 160 };

const defaultId = () => (globalThis.crypto?.randomUUID?.() ?? `${Date.now()}-${Math.random().toString(36).slice(2)}`);
const isCustom = (id) => typeof id === "string" && id.startsWith("c-");

export function emptyDeckEdits() {
  return { copies: {}, edits: {}, custom: [] };
}

export function parseCategory(category) {
  const rule = RULE.exec(category || "");
  if (rule) return { kind: "rule", who: rule[1], rounds: Number(rule[2]) };
  return { kind: category || "chaos" };
}

export function ruleCategory(who, rounds) {
  return `${who}_round_based:${rounds}`;
}

export function deckRows(base, edits) {
  const cards = [
    ...base.map((c) => (edits.edits[c.id] ? { ...c, ...edits.edits[c.id] } : c)),
    ...edits.custom,
  ];
  const copiesOf = (id) => edits.copies[id] ?? 1;
  const total = cards.reduce((n, c) => n + copiesOf(c.id), 0);
  return cards.map((card) => {
    const copies = copiesOf(card.id);
    return {
      card,
      copies,
      odds: total ? copies / total : 0,
      status: isCustom(card.id) ? "custom" : edits.edits[card.id] ? "edited" : "base",
    };
  });
}

export function buildDeck(rows) {
  return rows.flatMap((r) => Array(r.copies).fill(r.card));
}

export function setCopies(edits, id, n) {
  const copies = { ...edits.copies };
  const value = Math.max(0, Math.min(MAX_COPIES, Math.round(n)));
  if (value === 1) delete copies[id];
  else copies[id] = value;
  return { ...edits, copies };
}

export function validateCard({ title = "", description = "" }) {
  const errors = {};
  const t = title.trim();
  const d = description.trim();
  if (!t) errors.title = "Give it a title.";
  else if (t.length > LIMITS.title) errors.title = `Keep it to ${LIMITS.title} characters.`;
  if (!d) errors.description = "Say what to do.";
  else if (d.length > LIMITS.description) errors.description = `Keep it to ${LIMITS.description} characters.`;
  return errors;
}

export function saveCard(base, edits, draft, makeId = defaultId) {
  const fields = { title: draft.title.trim(), description: draft.description.trim(), category: draft.category };
  if (draft.id == null) {
    return { ...edits, custom: [...edits.custom, { id: `c-${makeId()}`, ...fields }] };
  }
  if (isCustom(draft.id)) {
    return { ...edits, custom: edits.custom.map((c) => (c.id === draft.id ? { ...c, ...fields } : c)) };
  }
  const original = base.find((c) => c.id === draft.id);
  const unchanged = original
    && original.title === fields.title
    && original.description === fields.description
    && (original.category || null) === fields.category;
  const next = { ...edits.edits };
  if (unchanged) delete next[draft.id];
  else next[draft.id] = fields;
  return { ...edits, edits: next };
}

export function restoreCard(edits, id) {
  const next = { ...edits.edits };
  delete next[id];
  return { ...edits, edits: next };
}

export function deleteCard(edits, id) {
  const copies = { ...edits.copies };
  delete copies[id];
  return { ...edits, copies, custom: edits.custom.filter((c) => c.id !== id) };
}

function matchesFilter(row, filter) {
  if (!filter || filter === "all") return true;
  if (filter === "mine") return row.status === "custom";
  const { kind } = parseCategory(row.card.category);
  return filter === "rules" ? kind === "rule" : kind === filter;
}

export function filterRows(rows, { query = "", filter = "all" } = {}) {
  const q = query.trim().toLowerCase();
  return rows.filter((r) => matchesFilter(r, filter)
    && (!q || `${r.card.title} ${r.card.description}`.toLowerCase().includes(q)));
}

export function shareOf(rows, filter) {
  const total = rows.reduce((n, r) => n + r.copies, 0);
  if (!total) return 0;
  return rows.filter((r) => matchesFilter(r, filter)).reduce((n, r) => n + r.copies, 0) / total;
}

export function formatOdds(p) {
  if (!p) return "—";
  const pct = p * 100;
  if (pct < 0.1) return "<0.1%";
  return `${pct < 10 ? pct.toFixed(1) : Math.round(pct)}%`;
}

import { useEffect, useMemo, useState } from "react";
import baseCards from "../data/cards.json";
import { buildDeck, deckRows, deleteCard, emptyDeckEdits, restoreCard, saveCard, setCopies } from "../lib/deck.js";

const KEY = "drunkgamer:deck";

function load() {
  try {
    const saved = JSON.parse(localStorage.getItem(KEY));
    if (saved?.copies && saved?.edits && Array.isArray(saved?.custom)) return saved;
  } catch { /* fall through to default */ }
  return emptyDeckEdits();
}

// This device's deck: copies per card, reworded built-ins and custom cards.
export function useDeck() {
  const [edits, setEdits] = useState(load);

  useEffect(() => {
    try { localStorage.setItem(KEY, JSON.stringify(edits)); } catch { /* storage unavailable */ }
  }, [edits]);

  const rows = useMemo(() => deckRows(baseCards, edits), [edits]);
  const cards = useMemo(() => buildDeck(rows), [rows]);

  return {
    rows,
    cards,
    unique: rows.filter((r) => r.copies > 0).length,
    setCopies: (id, n) => setEdits((e) => setCopies(e, id, n)),
    save: (draft) => setEdits(saveCard(baseCards, edits, draft)),
    restore: (id) => setEdits((e) => restoreCard(e, id)),
    remove: (id) => setEdits((e) => deleteCard(e, id)),
    reset: () => setEdits(emptyDeckEdits()),
  };
}

import { useEffect, useState } from "react";
import { addPlayer, dealCard, newGame, removePlayer } from "../lib/engine.js";

const KEY = "drunkgamer:game";

function load() {
  try {
    const saved = JSON.parse(localStorage.getItem(KEY));
    return saved?.players ? saved : null;
  } catch {
    return null;
  }
}

function save(game) {
  try {
    game ? localStorage.setItem(KEY, JSON.stringify(game)) : localStorage.removeItem(KEY);
  } catch { /* storage unavailable: game still works for this session */ }
}

// Game state lives on this device; a saved game resumes after a reload.
export function useGame() {
  const [game, setGame] = useState(load);
  const [error, setError] = useState("");

  useEffect(() => save(game), [game]);

  const screen = !game ? "start" : game.currentCard ? "game" : "lobby";

  // Compute outside the state updater so engine errors are caught here and
  // StrictMode's double-invoked updaters can't deal two different cards.
  const update = (fn) => {
    setError("");
    try {
      setGame(fn(game));
    } catch (e) {
      setError(e.message);
    }
  };

  return {
    screen,
    game,
    players: game?.players || [],
    error,
    start: () => update(() => newGame()),
    addPlayer: (name) => {
      if (!name.trim()) return false;
      update((g) => addPlayer(g, name));
      return true;
    },
    removePlayer: (id) => update((g) => removePlayer(g, id)),
    nextCard: (cards) => {
      if (!game?.players.length) return setError("Add at least one player first.");
      if (!cards.length) return setError("Your deck is empty. Add some cards in the DECK tab.");
      update((g) => dealCard(g, cards));
    },
    reset: () => update(() => null),
  };
}

                <span className="step__n" aria-live="polite" aria-label={`${copies} copies`}>×{copies}</span>
                <button className="step__b" onClick={() => deck.setCopies(card.id, copies + 1)} disabled={copies === MAX_COPIES}
                        aria-label={`More copies of ${card.title}`}>+</button>
              </span>
              <button className="drow__edit" onClick={() => setEditing(row)} aria-label={`Edit ${card.title}`}>✎</button>
            </li>
          );
        })}
      </ul>

      <div className="deck__reset">
        {confirmReset ? (
          <>
            <span className="vt deck__confirm">Put every card back to ×1, undo edits and delete your own cards?</span>
            <button className="btn btn--ghost deck__rbtn" onClick={() => setConfirmReset(false)}>KEEP</button>
            <button className="btn deck__rbtn deck__danger" onClick={() => { deck.reset(); setConfirmReset(false); }}>RESET</button>
          </>
        ) : (
          <button className="chip" onClick={() => setConfirmReset(true)}>RESET DECK TO DEFAULT</button>
        )}
      </div>

      {editing && (
        <CardEditor
          row={editing === "new" ? null : editing}
          onCancel={close}
          onSave={(draft) => { deck.save(draft); close(); }}
          onRestore={(id) => { deck.restore(id); close(); }}
          onDelete={(id) => { deck.remove(id); close(); }}
        />
      )}
    </div>
  );
}

import { useEffect, useRef, useState } from "react";
import { KINDS, LIMITS, parseCategory, ruleCategory, validateCard } from "../lib/deck.js";

const TYPES = [...KINDS, "rule"];

export default function CardEditor({ row, onCancel, onSave, onRestore, onDelete }) {
  const card = row?.card;
  const start = parseCategory(card?.category ?? "chaos");
  const [title, setTitle] = useState(card?.title ?? "");
  const [description, setDescription] = useState(card?.description ?? "");
  const [kind, setKind] = useState(start.kind);
  const [who, setWho] = useState(start.who ?? "person");
  const [rounds, setRounds] = useState(start.rounds ?? 1);
  const [errors, setErrors] = useState({});
  const titleRef = useRef(null);

  useEffect(() => { titleRef.current?.focus(); }, []);
  useEffect(() => {
    const onKey = (e) => { if (e.key === "Escape") onCancel(); };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [onCancel]);

  function submit(e) {
    e.preventDefault();
    const found = validateCard({ title, description });
    setErrors(found);
    if (Object.keys(found).length) return;
    onSave({ id: card?.id, title, description, category: kind === "rule" ? ruleCategory(who, rounds) : kind });
  }

  const choice = (value, current, set, label) => (
    <button type="button" key={value} className={"chip" + (current === value ? " is-on" : "")}
            aria-pressed={current === value} onClick={() => set(value)}>{label}</button>
  );

  return (
    <div className="modal" onClick={onCancel}>
      <form className="box editor" role="dialog" aria-modal="true" aria-labelledby="editor-h"
            onClick={(e) => e.stopPropagation()} onSubmit={submit} noValidate>
        <h2 id="editor-h" className="kick editor__h">{card ? "EDIT CARD" : "NEW CARD"}</h2>

        <label className="editor__lab" htmlFor="ed-title">TITLE</label>
        <input id="ed-title" ref={titleRef} className="input" value={title} maxLength={LIMITS.title}
               autoComplete="off" aria-invalid={Boolean(errors.title)} onChange={(e) => setTitle(e.target.value)} />
        {errors.title && <p className="editor__err">{errors.title}</p>}

        <label className="editor__lab" htmlFor="ed-desc">WHAT TO DO <span className="editor__count">{description.length}/{LIMITS.description}</span></label>
        <textarea id="ed-desc" className="input editor__desc" rows={3} value={description} maxLength={LIMITS.description}
                  aria-invalid={Boolean(errors.description)} onChange={(e) => setDescription(e.target.value)} />
        {errors.description && <p className="editor__err">{errors.description}</p>}

        <span className="editor__lab">TYPE</span>
        <div className="editor__chips">{TYPES.map((t) => choice(t, kind, setKind, t.toUpperCase()))}</div>

        {kind === "rule" && (
          <>
            <span className="editor__lab">WHO</span>
            <div className="editor__chips">
              {choice("person", who, setWho, "ONE PLAYER")}
              {choice("group", who, setWho, "EVERYONE")}
            </div>
            <span className="editor__lab">HOW MANY ROUNDS</span>
            <div className="editor__chips">{[1, 2, 3, 4].map((n) => choice(n, rounds, setRounds, String(n)))}</div>
          </>
        )}

        <div className="editor__acts">
          <button type="button" className="btn btn--ghost" onClick={onCancel}>CANCEL</button>
          {row?.status === "edited" && <button type="button" className="btn btn--ghost" onClick={() => onRestore(card.id)}>RESTORE</button>}
          {row?.status === "custom" && <button type="button" className="btn btn--ghost editor__del" onClick={() => onDelete(card.id)}>DELETE</button>}
          <button className="btn editor__save">SAVE</button>
        </div>
      </form>
    </div>
  );
}
